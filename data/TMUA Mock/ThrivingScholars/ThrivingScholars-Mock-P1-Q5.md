$$
\mathbf{A} \quad pa = qb
$$

$$
\mathbf{B} \quad p - a = q - b
$$

$$
\mathbf{C} \quad p - a < q - b
$$

$$
\mathbf{D} \quad p + q = a + b
$$

$$
\mathbf{E} \quad p - a > q - b
$$
